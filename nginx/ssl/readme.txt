🔍 Разбор КАЖДОЙ СТРОКИ openssl.cnf

[ req ]
Главная секция запроса сертификата

default_bits = 2048
🔐 Длина ключа
2048 — золотой стандарт, безопасно

prompt = no
❌ Не спрашивать вопросы в терминале
✔️ Всё берётся из файла

default_md = sha256
🔑 Алгоритм хеша
SHA-256 — современный стандарт

req_extensions = req_ext
👉 Говорим OpenSSL:
используй расширения из секции [req_ext]

distinguished_name = dn
👉 Основная информация о сертификате — в [dn]

[ dn ] — кто мы такие

C = TM
Страна (не критично)

ST = Ahal
L = Ashgabat
Регион / город (формальность)

O = Local Network
OU = Dev
Организация / подразделение

CN = 192.168.25.74
⚠️ Общее имя
Сейчас браузеры на него почти не смотрят
Но пусть совпадает с IP

[ req_ext ]
subjectAltName = @alt_names
🔥 КЛЮЧЕВАЯ СТРОКА
👉 Говорит:

IP и домены бери из секции [alt_names]
